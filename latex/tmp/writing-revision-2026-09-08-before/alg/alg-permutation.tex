\begin{algorithm}[!t]
\caption{Packed permutation proof}
\label{alg:permutation}
\KwIn{
\(N=m\kappa\), where \(m,\kappa\in2^{\mathbb N}\);
\(\bm a,\bm b\in\Field^N\)
}
\KwOut{
Reduced \(\ext{v}_a,\ext{v}_b\);
MLE points
\(\ext{\bm y}_a,\ext{\bm y}_b
\in\ExtensionField^{\log_2N}\);
acceptance bit \(\mathsf{ok}\)
}

After the commitments to \(\bm a\) and \(\bm b\) are fixed,
\(\VtoP\):
\(\ext z\ChooseRandom\ExtensionField\)\;

\ForEach{\(\bm d\in\{\bm a,\bm b\}\)}{
    Partition \(\bm d\) into \(\kappa\) blocks of length \(m\)
    and set
    \(
        \bm D^{(\bm d)}
        \leftarrow
        \concat{
            \bm d^{(1)},\ldots,\bm d^{(\kappa)}
        }
        \in\Field^{m\times\kappa}.
    \)

    \(\Prover\) constructs the column-product tree
    \(
        \bm C^{(\bm d)}_0(Z)
        \leftarrow
        Z\bm 1_{m\times\kappa}
        +
        \bm D^{(\bm d)}
    \)\;

    \For{\(t=0,\ldots,\log_2\kappa-1\)}{
        \(
            \bm C^{(\bm d)}_{t+1}(Z)
            \leftarrow
            \bm C^{(\bm d)}_{t,L}(Z)
            \circ
            \bm C^{(\bm d)}_{t,R}(Z).
        \)
    }

    \(\Prover\) evaluates the terminal column
    \(
        \ext{\bm c}^{(\bm d)}_0
        \leftarrow
        \bm C^{(\bm d)}_{\log_2\kappa}(\ext z)
        \in\ExtensionField^m.
    \)

    \For{\(i=0,\ldots,\log_2m-1\)}{
        \(
            \ext{\bm c}^{(\bm d)}_{i+1}
            \leftarrow
            \ext{\bm c}^{(\bm d)}_{i,L}
            \circ
            \ext{\bm c}^{(\bm d)}_{i,R}.
        \)
    }

    Set
    \(
        \Theta_{\bm d}
        \leftarrow
        \ext c^{(\bm d)}_{\log_2m}.
    \)
}

\(\PtoV\):
\(\Theta_{\bm a},\Theta_{\bm b}\)\;

\(\Verifier\) checks
\(
    \mathsf{ok}_{\mathsf{perm}}
    \leftarrow
    \left(
        \Theta_{\bm a}
        \EqOrNot
        \Theta_{\bm b}
    \right).
\)

\ForEach{\(\bm d\in\{\bm a,\bm b\}\)}{
    Set
    \(
        \ext{s}^{(\bm d)}_{\log_2m}
        \leftarrow
        \Theta_{\bm d}
    \)
    and
    \(
        \ext{\bm\xi}^{(\bm d)}_{\log_2m}
        \leftarrow(1).
    \)

    \tcp{Unfold the row-product tree}
    \For{\(i=\log_2m-1,\ldots,0\)}{
        Interpret the current claim as
        \(
            \ext{s}^{(\bm d)}_{i+1}
            \EqOrNot
            \left\langle
                \ext{\bm c}^{(\bm d)}_{i,L},
                \ext{\bm c}^{(\bm d)}_{i,R}
                \circ
                \ext{\bm\xi}^{(\bm d)}_{i+1}
            \right\rangle.
        \)

        \(\Prover\) and \(\Verifier\) apply the inner-product
        reduction of Fact~\ref{fact:ip} and randomly combine its
        terminal claims, obtaining
        \(
            \ext{s}^{(\bm d)}_i
            \EqOrNot
            \left\langle
                \ext{\bm\xi}^{(\bm d)}_i,
                \ext{\bm c}^{(\bm d)}_i
            \right\rangle
        \)
        and the acceptance bit
        \(\mathsf{ok}^{(\bm d)}_{\mathsf{row},i}\), where
        \(\ext{\bm\xi}^{(\bm d)}_i\) is tensor-structured\;
    }

    Set
    \(
        \ext{q}^{(\bm d)}_{\log_2\kappa}
        \leftarrow
        \ext{s}^{(\bm d)}_0
    \)
    and
    \(
        \ext{\bm\zeta}^{(\bm d)}_{\log_2\kappa}
        \leftarrow
        \ext{\bm\xi}^{(\bm d)}_0.
    \)

    \tcp{Unfold the column-product tree}
    \For{\(t=\log_2\kappa-1,\ldots,0\)}{
        Interpret the current claim as
        \(
            \ext{q}^{(\bm d)}_{t+1}
            \EqOrNot
            \left\langle
                \ext{\bm\zeta}^{(\bm d)}_{t+1},
                \operatorname{vec}
                \bigl(
                    \bm C^{(\bm d)}_{t,L}(\ext z)
                \bigr)
                \circ
                \operatorname{vec}
                \bigl(
                    \bm C^{(\bm d)}_{t,R}(\ext z)
                \bigr)
            \right\rangle.
        \)

        Apply the coefficient-packed claim reduction of
        Algorithm~\ref{alg:hadamard}, with its evaluated
        Hadamard claim written as an inner product and reduced
        using Fact~\ref{fact:ip}, obtaining
        \(
            \ext{q}^{(\bm d)}_t
            \EqOrNot
            \left\langle
                \ext{\bm\zeta}^{(\bm d)}_t,
                \operatorname{vec}
                \bigl(
                    \bm C^{(\bm d)}_t(\ext z)
                \bigr)
            \right\rangle
        \)
        and the acceptance bit
        \(\mathsf{ok}^{(\bm d)}_{\mathsf{col},t}\), where
        \(\ext{\bm\zeta}^{(\bm d)}_t\) is tensor-structured\;
    }

    Write
    \(
        \ext{\bm\zeta}^{(\bm d)}_0
        =
        \tensorspan{\ext{\bm y}_{\bm d}}
    \)
    for
    \(
        \ext{\bm y}_{\bm d}
        \in\ExtensionField^{\log_2N}.
    \)

    \(\Prover\) and \(\Verifier\) compute
    \(
        \ext{v}_{\bm d}
        \leftarrow
        \ext{q}^{(\bm d)}_0
        -
        \ext z
        \left\langle
            \ext{\bm\zeta}^{(\bm d)}_0,
            \bm 1_N
        \right\rangle.
    \)

    The resulting terminal claim is
    \(
        \ext{v}_{\bm d}
        \EqOrNot
        \left\langle
            \bm d,
            \tensorspan{\ext{\bm y}_{\bm d}}
        \right\rangle.
    \)
}

Set
\(
    \ext{v}_a\leftarrow\ext{v}_{\bm a}
\),
\(
    \ext{v}_b\leftarrow\ext{v}_{\bm b}
\),
\(
    \ext{\bm y}_a\leftarrow\ext{\bm y}_{\bm a}
\),
and
\(
    \ext{\bm y}_b\leftarrow\ext{\bm y}_{\bm b}.
\)

Set
\(
    \mathsf{ok}
    \leftarrow
    \mathsf{ok}_{\mathsf{perm}}
    \wedge
    \bigwedge_{\bm d\in\{\bm a,\bm b\}}
    \left(
        \bigwedge_{i=0}^{\log_2m-1}
        \mathsf{ok}^{(\bm d)}_{\mathsf{row},i}
    \right)
    \wedge
    \bigwedge_{\bm d\in\{\bm a,\bm b\}}
    \left(
        \bigwedge_{t=0}^{\log_2\kappa-1}
        \mathsf{ok}^{(\bm d)}_{\mathsf{col},t}
    \right).
\)

\Return
\(
    (
        \mathsf{ok},
        \ext{v}_a,
        \ext{v}_b,
        \ext{\bm y}_a,
        \ext{\bm y}_b
    )
\)\;
\end{algorithm}
