\begin{algorithm}[!t]
\caption{Packed Hadamard-product proof}
\label{alg:hadamard}
\KwIn{
\(N=m\kappa\), where \(m,\kappa\in2^{\mathbb N}\);
\(\ext{c}\in\ExtensionField\);
\(\bm a,\bm b\in\Field^N\);
\(\ext{\bm\xi}
=\ext{\bm r}\kronecker\ext{\bm l}\),
where
\(\ext{\bm r}\in\ExtensionField^\kappa\) and
\(\ext{\bm l}
=\tensorspan{\ext{\bm z}_l}\in\ExtensionField^m\)
}
\KwOut{
Reduced \(\ext{v}_1,\ext{v}_2\);
MLE points
\(\ext{\bm y}_a,\ext{\bm y}_b
\in\ExtensionField^{\log_2N}\);
acceptance bit \(\mathsf{ok}\)
}

Partition \(\bm a\) and \(\bm b\) into \(\kappa\) blocks of
length \(m\), and set
\[
    \bm A
    \leftarrow
    \concat{
        \bm a^{(1)},\ldots,\bm a^{(\kappa)}
    },
    \qquad
    \bm B
    \leftarrow
    \concat{
        \bm b^{(1)},\ldots,\bm b^{(\kappa)}
    }
    \in\Field^{m\times\kappa}.
\]

\(\Prover\) computes the column claims
\(
    \ext{\bm c}^{\top}
    \leftarrow
    \ext{\bm l}^{\top}
    (\bm A\circ\bm B)
    \in\ExtensionField^\kappa.
\)

Using the Lagrange basis from
Definition~\ref{def:lagrange}, \(\Prover\) computes:\\
\Indp
\(    \bm g_A(X)
    \leftarrow
    \sum_{j=1}^{\kappa}
    \bm a^{(j)}\psi_j(X)
\);~
\(
    \bm g_B(X)
    \leftarrow
    \sum_{j=1}^{\kappa}
    \bm b^{(j)}\psi_j(X)
\)\;
\(
    g_C(X)
    \leftarrow
    \sum_{j=1}^{\kappa}
    \ext{c}_j\psi_j(X)
\)\;

Construct the residual polynomial
\(
    f(X)
    \leftarrow
    \left\langle
        \ext{\bm l},
        \bm g_A(X)\circ\bm g_B(X)
    \right\rangle
    -
    g_C(X)
\)
and its quotient
\(
    h(X)
    \leftarrow
    \frac{f(X)}{\Psi(X)}
    \in\ExtensionField[X].
\)
\Indm

\(\PtoV\):
\(\ext{\bm c}\) and the \(\kappa-1\) coefficients of \(h(X)\)\;

\(\Verifier\) computes
\(
    \mathsf{ok}_0
    \leftarrow
    \left(
        \ext{c}
        \EqOrNot
        \left\langle
            \ext{\bm c},\ext{\bm r}
        \right\rangle
    \right).
\)

\(\VtoP\):
\(
    \ext{x}
    \ChooseRandom
    \ExtensionField\setminus\mathcal H.
\)

\(\Prover\) and \(\Verifier\) compute the Lagrange packing vector
\(
    \ext{\bm\rho}
    \leftarrow
    \bigl(
        \psi_1(\ext{x}),
        \ldots,
        \psi_\kappa(\ext{x})
    \bigr)^\top
    \in\ExtensionField^\kappa.
\)

\(\Prover\) computes:
\(
    \ext{\bm a}
    \leftarrow
    \bm g_A(\ext{x})
    =
    \bm A\ext{\bm\rho}
    \in\ExtensionField^m
\);~
\(
    \ext{\bm b}
    \leftarrow
    \bm g_B(\ext{x})
    =
    \bm B\ext{\bm\rho}
    \in\ExtensionField^m
\)\;

\(\Verifier\) computes:
\(
    \ext{c}_x
    \leftarrow
    g_C(\ext{x})
    =
    \left\langle
        \ext{\bm c},\ext{\bm\rho}
    \right\rangle
\);~
\(
    \ext{d}
    \leftarrow
    \ext{c}_x
    +
    \Psi(\ext{x})h(\ext{x})
\)\;

\(\Prover\) and \(\Verifier\) apply the inner-product reduction
of Fact~\ref{fact:ip} to
\(
    \ext{d}
    \EqOrNot
    \left\langle
        \ext{\bm a},
        \ext{\bm b}\circ\ext{\bm l}
    \right\rangle,
\)
obtaining the reduced scalars
\(\ext{u}_1,\ext{u}_2\), the challenge point
\(
    \ext{\bm y}'
    \in\ExtensionField^{\log_2m},
\)
and the acceptance bit \(\mathsf{ok}_{\mathsf{IP}}\)\;

\(\Prover\) computes, for every \(j\in[\kappa]\),
\(
    \ext{v}_{a,j}
    \leftarrow
    \left\langle
        \bm a^{(j)},
        \tensorspan{\ext{\bm y}'}
    \right\rangle
\);
\(
    \ext{v}_{b,j}
    \leftarrow
    \left\langle
        \bm b^{(j)},
        \ext{\bm l}
        \circ
        \tensorspan{(\ext{\bm y}')}
    \right\rangle
\)\;

Set
\(
    \ext{\bm v}_a
    \leftarrow
    \bigl(
        \ext{v}_{a,1},\ldots,\ext{v}_{a,\kappa}
    \bigr)^\top
\);~
\(
    \ext{\bm v}_b
    \leftarrow
    \bigl(
        \ext{v}_{b,1},\ldots,\ext{v}_{b,\kappa}
    \bigr)^\top.
\)

\(\PtoV\):
\(\ext{\bm v}_a,\ext{\bm v}_b\)\;

\(\Verifier\) checks:
\(
    \mathsf{ok}_{\mathsf{Lag}}
    \leftarrow
    \left(
        \ext{u}_1
        \EqOrNot
        \left\langle
            \ext{\bm v}_a,\ext{\bm\rho}
        \right\rangle
    \right)
    \wedge
    \left(
        \ext{u}_2
        \EqOrNot
        \left\langle
            \ext{\bm v}_b,\ext{\bm\rho}
        \right\rangle
    \right).
\)

\(\VtoP\):
\(
    \ext{\bm{y}}''
    \ChooseRandom
    \ExtensionField^{\log_2 \kappa}.
\)

\(\Prover\) and \(\Verifier\) compute:
\(
    \ext{v}_1
    \leftarrow
    \left\langle
        \ext{\bm v}_a,
        \tensorspan{\ext{\bm y}''}
    \right\rangle
\);~
\(
    \ext{v}_2
    \leftarrow
    \left\langle
        \ext{\bm v}_b,
        \tensorspan{\ext{\bm y}''}
    \right\rangle
\)\;

They then construct the final MLE points:\\
\Indp
\(
    \ext{\bm y}_a
    \leftarrow
    \ext{\bm y}''
    \mathbin{\|}
    \ext{\bm y}'\in\ExtensionField^{\log_2N}
\);~
\(
    \ext{\bm y}_b
    \leftarrow
    \ext{\bm y}''
    \mathbin{\|}
    \left(
        \ext{\bm z}_l
        \circ
        \ext{\bm y}'
    \right)\in\ExtensionField^{\log_2N}
\)\;
\Indm

The resulting terminal claims are
\(
    \ext{v}_1
    \EqOrNot
    \left\langle
        \bm a,
        \tensorspan{\ext{\bm y}_a}
    \right\rangle
    \wedge
    \ext{v}_2
    \EqOrNot
    \left\langle
        \bm b,
        \tensorspan{\ext{\bm y}_b}
    \right\rangle.
\)\;

Set
\(
    \mathsf{ok}
    \leftarrow
    \mathsf{ok}_0
    \wedge
    \mathsf{ok}_{\mathsf{IP}}
    \wedge
    \mathsf{ok}_{\mathsf{Lag}}.
\)

\Return
\(
    (
        \mathsf{ok},
        \ext{v}_1,
        \ext{v}_2,
        \ext{\bm y}_a,
        \ext{\bm y}_b
    )
\)\;
\end{algorithm}
