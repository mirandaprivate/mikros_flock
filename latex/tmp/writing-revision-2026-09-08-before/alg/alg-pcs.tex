\begin{algorithm}[!t]
\caption{Packed opening protocol}
\label{alg:packed-pcs}
\KwPubIn{
\(\mathsf{pp}_{\pc},\mathsf{cm}\);
\(\Field\subset\ExtensionField\) with basis \(\ext{\bm\beta}\);
\(N=m\kappa\), \(k\) as in Eq.~\eqref{eq:pcs-repetition-number};
\(\ext u\in\ExtensionField\),
\(\ext{\bm\rho}\in\ExtensionField^\kappa\), and
\(\ext{\bm\chi}=\bigotimes_{j=1}^{\log_2m}
\ext{\bm\chi}^{(j)}\), given by factors
\(\ext{\bm\chi}^{(j)}\in\ExtensionField^2\)
}
\KwPriIn{
\(\Prover\): \(\bm A\in\Field^{m\times\kappa}\) and
commitment randomness \(\rho_{\pc}\)
}
\KwOut{Acceptance bit \(\mathsf{ok}\)}

\(\Verifier\) rejects malformed messages and returns \(0\)
on any failed check or inner-product reduction\;
\(\Prover\) sets
\(\ext{\bm a}\leftarrow\bm A\ext{\bm\beta}\) and extracts
\(\bm\Xi\in\Field^{m\times\kappa}\) from
\(\ext{\bm\chi}=\bm\Xi\ext{\bm\beta}\)\;
\(\Prover\) computes \(\bm V\leftarrow\bm A^\top\bm\Xi\)\;
\(\PtoV\): \(\ext{\bm v}\leftarrow\bm V\ext{\bm\beta}
\in\ExtensionField^\kappa\)\;
\(\Verifier\) checks
\(\ext u\EqOrNot\ext{\bm\rho}^{\top}\ext{\bm v}\)\;
\(\Verifier\) recovers \(\bm V\in\Field^{\kappa\times\kappa}\)
from \(\ext{\bm v}=\bm V\ext{\bm\beta}\)\;
\(\VtoP\): \(\ext{\bm r}\ChooseRandom\ExtensionField^\kappa\)\;
\(\Verifier\) computes
\(\ext c\leftarrow\ext{\bm\beta}^{\top}\bm V\ext{\bm r}\)\;
\(\Prover\) computes the length-\(m\) operand
\(\bm\Xi\ext{\bm r}\in\ExtensionField^m\)\;
\(\Prover,\Verifier\) apply Fact~\ref{fact:ip} over
\(\ExtensionField\) to
\(\ext c\EqOrNot\langle\ext{\bm a},\bm\Xi\ext{\bm r}\rangle\),
obtaining terminal values \(\ext w_{\mathsf{com}},\ext w\)
and point \(\ext{\bm y}\in\ExtensionField^{\log_2m}\)\;
Set \(\ext{\bm\xi}\leftarrow\tensorspan{\ext{\bm y}}\)\;
\(\Prover\) extracts \(\bm U\in\Field^{m\times\kappa}\)
from \(\ext{\bm\xi}=\bm U\ext{\bm\beta}\)\;
\(\Prover\) computes \(\bm W\leftarrow\bm U^\top\bm\Xi\)\;
\(\PtoV\): \(\bm W\in\Field^{\kappa\times\kappa}\)\;
\(\Verifier\) checks
\(\ext w\EqOrNot\ext{\bm\beta}^{\top}\bm W\ext{\bm r}\)\;

\ForEach{\(i\in[k]\)}{
    \(\VtoP\): fresh independent
    \(\bm s_{(i)},\bm t_{(i)}\ChooseRandom\Field^\kappa\)\;
    \(\Prover\) computes
    \(\bm U\bm s_{(i)},\bm\Xi\bm t_{(i)}\in\Field^m\)\;
    \(\Prover,\Verifier\) apply Fact~\ref{fact:ip} over
    \(\Field\) to
    \(\bm s_{(i)}^\top\bm W\bm t_{(i)}
      \EqOrNot\langle\bm U\bm s_{(i)},\bm\Xi\bm t_{(i)}\rangle\),
    using fresh base-field challenges\;
    Let \(v_{(i),1},v_{(i),2}\in\Field\) and
    \(\bm y_{(i)}\in\Field^{\log_2m}\) be the terminal
    values and point; set
    \(\bm l_{(i)}\leftarrow\tensorspan{\bm y_{(i)}}\)\;
    \(\Verifier\) computes
    \(\langle\bm l_{(i)},\ext{\bm\xi}\rangle\) and
    \(\langle\bm l_{(i)},\ext{\bm\chi}\rangle\)
    from their tensor factors\;
    \(\Verifier\) decodes their basis coordinates as the rows
    \(\bm l_{(i)}^\top\bm U\) and
    \(\bm l_{(i)}^\top\bm\Xi\), respectively\;
    \(\Verifier\) checks
    \(v_{(i),1}\EqOrNot
      (\bm l_{(i)}^\top\bm U)\bm s_{(i)}\) and
    \(v_{(i),2}\EqOrNot
      (\bm l_{(i)}^\top\bm\Xi)\bm t_{(i)}\)\;
}

\(\mathsf{ok}\leftarrow
\pc.\Open\!\left(
    \mathsf{pp}_{\pc},\mathsf{cm},\ext w_{\mathsf{com}},
    \ext{\bm y}:\ext{\bm a},\rho_{\pc}
\right)\)\;
\Return \(\mathsf{ok}\)\;
\end{algorithm}
