\section{Definitions of Completeness, Soundness and Knowledge Soundness}
\label{appx:def}

\subsection{Concepts of Soundness}
\label{appx:soundness}
\begin{definition}[Proof/Argument/Argument of Knowledge]
\label{def:proofArgumentAoK}
A proof, an argument, and an argument of knowledge,
defined by the $(\Setup, \Prover, \Verifier)$ tuple below,
should satisfy 
soundness,
computational soundness, and 
knowledge soundness,
respectively.
They all possess perfect completeness.
\begin{itemize}
	\item $\Setup$:
	Given a security parameter $1^\lambda$, outputs public parameters $\pp$.
	\item Prover $\Prover$:
	Given the public parameters $\pp$, the statement $u$, outputs a proof transcript $\pi$ for a language $\Language$.
	Alternatively, given $\pp$, a statement $u$, and a private witness $w$, the prover outputs a transcript $\pi$ for a relation $\Relation$.
	\item Verifier $\Verifier$:
	Given $\pp$, statement $u$, and transcript $\pi$, outputs {\sf true} or {\sf false}.
\end{itemize}
Let $\PVinteract$ denote the interaction between $\Prover$ and $\Verifier$.
\end{definition}

\begin{definition}[Perfect Completeness]
	\label{def:perfect-completeness}
The protocol $(\Setup,\Prover,\Verifier)$ for a relation \(\Relation\)
has perfect completeness if, for every \((u,w)\in\Relation\),
\[
    \Pr\left[
        \Verifier(\pp,u,\pi)=\TRUE
        \;\middle|\;
        \pp\leftarrow\Setup(1^\lambda),\;
        \pi\leftarrow\Prover(\pp,u,w)
    \right]
    =1,
\]
where the probability is over the randomness of
\(\Setup\), \(\Prover\), and \(\Verifier\).
\end{definition} 

\begin{definition}[Soundness~\cite{iacr/CanettiCHLRR18}]
	\label{def:soundness}
	$(\Setup, \Prover, \Verifier)$ for a language $\Language$ / a relation $\mathcal{R}$ has a soundness error $\kappa^\mathsf{s}(u)\colon \mathbb{N} \mapsto [0, 1]$ if for any unbounded interactive $\Prover^*$, the probability 
	below
	%that $\Prover^*$ outputs an accepting transcript 
	is bounded by $\kappa^\mathsf{s}(u)$.
	If $\kappa^\mathsf{s}(u)$ is negligible,
	the protocol is sound.
	\begin{equation*}
	% \label{eq:soundnessExpectedTime}
	\Pr
		\left(
			\left.
			\begin{aligned}
				& \Verifier(\pp, u, \pi) = \TRUE \\
			\end{aligned}
			\right.
			\left|
			\begin{aligned}
				& u \notin \Language~/~ \Relation;~ \pi \leftarrow (\Prover^* \interact \Verifier) (\pp, u) \\
			\end{aligned}
			\right.
		\right)
	\le \kappa^\mathsf{s}(u).
	\end{equation*}
\end{definition}

An argument possesses computational soundness,
weaker than soundness:
\begin{definition}[Computational Soundness]
	\label{def:ComputationalSoundness}
	A protocol $(\Setup,\Prover,\Verifier)$ for the language $\Language$
    / a relation $\mathcal{R}$ has
	%is computational sound with 
	a computational soundness error $\kappa^\mathsf{cs}(u)\colon \mathbb{N} \mapsto [0, 1]$ if for any PPT adversary $\Prover^*$, the probability below
	%that $\Prover^*$ outputs an accepting transcript 
	is bounded by $\kappa^\mathsf{cs}(u)$.
	If $\kappa^\mathsf{cs}(u)$ is negligible, the protocol is computationally sound.
	\begin{equation*}
	% \label{eq:soundnessExpectedTime}
	\Pr
		\left(
			\left.
			\begin{aligned}
				& \Verifier(\pp, u, \pi) = \TRUE \\
			\end{aligned}
			\right.
			\left|
			\begin{aligned}
				& u \notin \Language~/~\Relation;~ \pi \leftarrow (\Prover^* \interact \Verifier)(\pp, u) \\
			\end{aligned}
			\right.
		\right)
	\le \kappa^\mathsf{cs}(u).
	\end{equation*}
\end{definition}
In interactive proofs, by definition, $\kappa^\mathsf{cs}(u) \le \kappa^\mathsf{s}(u)$.

An AoK requires knowledge soundness~\cite{crypto/AttemaCK21}:
\begin{definition}[Knowledge Soundness]
	\label{def:knowledgeSoundness}
	A protocol $(\Setup, \Prover, \Verifier)$ for the relation $\Relation$ is knowledge sound with a knowledge soundness error of $\kappa^\mathsf{ks}(u)\colon \mathbb{N} \mapsto [0, 1]$ if there exists an algorithm $\Extractor$, called a knowledge extractor, with the following characteristics:
	Given input $u$ and black-box oracle access to a (potentially dishonest) prover $\Prover^*$, the extractor $\Extractor$ runs in an expected number of steps that is polynomial in $|u|$ (where queries to $\Prover^*$ count as a single step) and outputs a witness $ (u, w) \in \Relation$ with probability:
	\begin{equation*}
	% \label{eq:soundnessExpectedTime}
	\Pr((u, \Extractor(\Prover^*, u) ) \in \Relation )
	\ge
	\epsilon(\Prover^*, u) - \kappa^\mathsf{ks}(u),
	\end{equation*}
	where $\epsilon(\Prover^*, u)$ represents the probability that $\Prover^*$ generates an accepting transcript.
	The protocol is considered knowledge sound if $\kappa^\mathsf{ks}(u)$ is negligible.
\end{definition}

If a protocol is knowledge-sound with knowledge-soundness error
$\kappa^\mathsf{ks}(u)$, then it is also computationally sound with
error $\kappa^\mathsf{cs}(u) \le \kappa^\mathsf{ks}(u)$.
Otherwise, for some input $u \notin \Relation$ and a PPT adversary $\Prover^*$, we would have:
\[
	\Pr
	\left(
		\left.
		\begin{aligned}
			& \Verifier(\pp, u, \pi) = \TRUE \\
		\end{aligned}
		\right.
		\left|
		\begin{aligned}
			& u \notin \mathcal{R};~ \pi \leftarrow (\Prover^* \interact \Verifier)(\pp, u) \\
		\end{aligned}
		\right.
	\right)
	> \kappa^\mathsf{ks}(u).
\]
In this case, for such an input $u$, the extractor $\Extractor$ must be able to extract a witness $w$ with positive probability such that $(u, w) \in \Relation$,
\ie, $\exists w~\suchthat~(u, w) \in \Relation$
This contradicts the assumption that $u \notin \Relation$.


\subsection{Round-by-Round Soundness}
\label{appx:rbyDef}

\begin{definition}[Round-by-Round Soundness~\cite{iacr/CanettiCHLRR18}]
	\label{def:roundByRoundSoundness}
	Let $\Pi = (\Setup, \Prover, \Verifier)$ be a $(2 \nu + 1)$-message public coin interactive proof system for a relation $\Relation$.
	For any statement $u \in \{0, 1\}^*$, and any prefix $\pi_j$ of a protocol transcript.
	Let $\mathsf{H}(u, \pi_j)$ denote the distribution of the next message of $\Verifier$ when the transcript so far is $\pi_j = (\alpha_0, x_1, \alpha_1, \ldots, \alpha_{j - 1}, x_j, \alpha_j), j \in [0, \nu]$ contains the challenges and responses of the in the initial $2j + 1$ moves.
	$\Pi$ has round-by-round soundness error $\{ \kappa^\mathsf{rbr}_{j}\}_{j \in [0, \nu]}$ if there exists a series of deterministic (not necessarily efficiently computable) functions $\{\mathsf{State}_j\}_{j \in [0, \nu]}$ that takes as input a statement $u$ and a transcript prefix $\pi_j$ and outputs either $\mathsf{accept}$ or $\mathsf{reject}$ such that the following properties hold:
		\begin{itemize}
			\item If $u \notin \Language$, then $\mathsf{State}_0(u, \emptyset) = \mathsf{reject}$, where $\emptyset$ denotes the empty transcript.
			\item If $\mathsf{State}_{j - 1}(u, \pi_{j-1}) = \mathsf{reject}$ for a transcript prefix $\pi_{j - 1}$, then for every potential prover's message $\alpha_{j}$ and verifier's challenge $x_{j}$, it holds that:
			\[
				\Pr\!\!
				\left(
					\mathsf{State}_j(u, \pi_j) = \mathsf{accept~}\!\!
					\left|~\!
						\begin{aligned}
							&\!\mathsf{State}_{j - 1}(u, \pi_{j - 1}) = \mathsf{reject}; \\
							& x_j \leftarrow \mathsf{H}(u, \pi_{j - 1}); \pi_j \leftarrow (\pi_{j - 1} \| x_j \| \alpha_j)\!
						\!\end{aligned}
					\right.
				\right)\!
				\!\le\!\kappa_j^\mathsf{rbr}.
			\]
			\item For any full transcript $\pi$ for which the verifier halts, if $\mathsf{State}_\nu(u, \pi) = \mathsf{reject}$ then $\Verifier(u, \pi) = \FALSE$.
		\end{itemize}
\end{definition}

Below lemma connects the round-by-round and overall soundness errors.
\begin{lemma}[Round-by-Round Soundness Implies Soundness~\hspace{-1.58pt}\textup{\cite[Proposition~5.4]{iacr/CanettiCHLRR18}}]
\label{lemma:rbr}
	Let $\Pi$ be a $(2 \nu + 1)$-move interactive proof with round-by-round soundness error $\kappa_j^\mathsf{rbr}, j \in [\nu]$.
	Then, $\Pi$ has a soundness error $\sum_{j = 1}^\nu \kappa_j^\mathsf{rbr}$.
\end{lemma}
