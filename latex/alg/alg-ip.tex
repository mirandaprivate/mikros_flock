\begin{algorithm}[!t]
\caption{Packed inner-product proof}
\label{alg:ip}
\KwIn{
\(N=m\kappa\), \(m\in2^{\mathbb N}\), \(\kappa\geq2\);
a fixed claim \(c\in\Field\) and fixed matrices
\(\bm A,\bm B\in\Field^{m\times\kappa}\)
}
\KwOut{
Terminal values \(\ext v_1,\ext v_2\);
column weights \(\ext{\bm\rho}_A,\ext{\bm\rho}_B\in\ExtensionField^\kappa\);
row MLE point \(\ext{\bm y}\in\ExtensionField^{\log_2m}\);
acceptance bit \(\mathsf{ok}\)
}

\tcp{Fix the corrections before the packing challenge}
\(\Prover\) defines
\(\bm g_A(X)\leftarrow\sum_{j=1}^{\kappa}\bm a^{(j)}X^{j-1}\)
and
\(\bm g_B(X)\leftarrow\sum_{j=1}^{\kappa}\bm b^{(j)}X^{-(j-1)}\)\;

\ForEach{\(\tau\in\{-(\kappa-1),\ldots,-1,1,\ldots,\kappa-1\}\)}{
    \(\Prover\) computes
    \(f_\tau\leftarrow
      \sum_{\substack{j,j'\in[\kappa]\\j-j'=\tau}}
      \langle\bm a^{(j)},\bm b^{(j')}\rangle\)\;
}
\(\Prover\) sets
\(h_+(X)\leftarrow\sum_{\tau=1}^{\kappa-1}f_\tau X^{\tau-1}\)
and
\(h_-(X)\leftarrow\sum_{\tau=1}^{\kappa-1}f_{-\tau}X^{\tau-1}\)\;
\(\PtoV\): the \(\kappa-1\) coefficients of each of \(h_+\) and \(h_-\)\;
\(\VtoP\): \(\ext x\ChooseRandom\ExtensionField^\times\)\;

\tcp{Reduce to a length-\(m\) extension-field inner product}
\(\Prover\) and \(\Verifier\) compute
\(\ext{\bm\rho}_A\leftarrow(1,\ext x,\ldots,\ext x^{\kappa-1})^\top\)
and
\(\ext{\bm\rho}_B\leftarrow(1,\ext x^{-1},\ldots,\ext x^{-(\kappa-1)})^\top\)\;
\(\Prover\) computes
\(\ext{\bm a}\leftarrow\bm A\ext{\bm\rho}_A=\bm g_A(\ext x)\)
and
\(\ext{\bm b}\leftarrow\bm B\ext{\bm\rho}_B=\bm g_B(\ext x)\)\;
\(\Verifier\) computes
\(\ext c\leftarrow
c+\ext x h_+(\ext x)+\ext x^{-1}h_-(\ext x^{-1})\)\;
\(\Prover\) and \(\Verifier\) apply Fact~\ref{fact:ip} to
\(\ext c\EqOrNot\langle\ext{\bm a},\ext{\bm b}\rangle\),
obtaining \(\mathsf{ok}\), \(\ext v_1,\ext v_2\), and
\(\ext{\bm y}\in\ExtensionField^{\log_2m}\)\;

\tcp{Output the two claims for packed opening}
Output
\(\ext v_1\EqOrNot
\tensorspan{\ext{\bm y}}^\top\bm A\ext{\bm\rho}_A\)
and
\(\ext v_2\EqOrNot
\tensorspan{\ext{\bm y}}^\top\bm B\ext{\bm\rho}_B\)\;
\Return{\((\mathsf{ok},\ext v_1,\ext v_2,
\ext{\bm\rho}_A,\ext{\bm\rho}_B,\ext{\bm y})\)}\;
\end{algorithm}
