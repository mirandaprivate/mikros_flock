\begin{algorithm}[!t]
\small
\caption{Packed permutation proof}
\label{alg:permutation}
\KwIn{
Vectors \(\bm a,\bm b\in\Field^N\) fixed by the input commitments, arranged as
\(\bm A,\bm B\in\Field^{m\times\kappa}\);
\(N=m\kappa\), with \(m,\kappa\) powers of two and \(m\geq\kappa\geq2\);
the interpolation basis \(\bm\psi\) and vanishing polynomial \(\Psi\)
}
\KwOut{
Two terminal packed claims
\(\ext v_0-\ext z=\ext{\bm l}_0^\top\bm D\ext{\bm r}_0\),
one for each \(\bm D\in\{\bm A,\bm B\}\), and an acceptance bit
}

\(\Verifier\) rejects malformed messages and failed scalar or sumcheck checks\;
Use nodes \(\mathcal H,\{\gamma'_j\}_{j=1}^{\kappa-1}\) in \(\Field\)
if \(q\geq2\kappa-1\); otherwise use
\(\Field\subseteq\mathbb K\subseteq\ExtensionField\)
and the addition and multiplication tables of Section~\ref{sec:hadamard}\;
After the commitments are fixed,
\(\VtoP:\ext z\ChooseRandom\ExtensionField\)\;

\ForEach{\(\bm D\in\{\bm A,\bm B\}\)}{
    \tcp{Construct the coefficient and extension-field trees}
    Construct \(\bm C_0(Z),\ldots,\bm C_{\log_2\kappa}(Z)\)
    over \(\Field[Z]\) as in
    Eq.~\eqref{eq:packed-permutation-product-tree}, starting from
    \(\bm C_0(Z)\leftarrow Z\bm1+\bm D\);
    retain only the current coefficient layer\;
    Evaluate and flatten
    \(\bm c_{\log_2\kappa}(\ext z)
      \leftarrow\operatorname{vec}(\bm C_{\log_2\kappa}(\ext z))
      \in\ExtensionField^m\)\;
    Complete the tree over \(\ExtensionField\), using
    \(\bm c_{\tau+1}(\ext z)
    \leftarrow\bm c_{\tau,U}(\ext z)\circ\bm c_{\tau,D}(\ext z)\)
    for \(\tau=\log_2\kappa,\ldots,\log_2N-1\)\;
    Retain this extension-field tree for \(\bm D\)\;
}
\(\PtoV:\ext\Theta\in\ExtensionField\), the claimed common root\;

\ForEach{\(\bm D\in\{\bm A,\bm B\}\)}{
    Use the extension-field tree associated with \(\bm D\)\;
    \tcp{Unfold the extension-field tree}
    Set \(\ext v\leftarrow\ext\Theta\) and
    \(\ext{\bm\xi}\leftarrow(1)\)\;
    \For{\(\tau=\log_2N-1,\ldots,\log_2\kappa\)}{
        Apply cubic weighted sumcheck to
        \(\ext v=\langle\ext{\bm\xi},
        \bm c_{\tau,U}(\ext z)\circ\bm c_{\tau,D}(\ext z)\rangle\),
        keeping the two child MLEs at the same point \(\ext{\bm y}\)\;
        After the child values \(\ext v_U,\ext v_D\) are fixed, sample
        \(\VtoP:\ext y_0\ChooseRandom\ExtensionField\) and set
        \(\ext v\leftarrow(1-\ext y_0)\ext v_U+\ext y_0\ext v_D\),
        \(\ext{\bm\xi}\leftarrow\tensorspan{\ext y_0\mathbin\|\ext{\bm y}}\)\;
    }
    Set \(\ext v_{\log_2\kappa}\leftarrow\ext v\) and factor
    \(\ext{\bm\xi}
      =\ext{\bm r}_{\log_2\kappa}\otimes\ext{\bm l}_{\log_2\kappa}\)\;

    \tcp{Unfold the coefficient tree, aggregating before evaluation}
    \For{\(\tau=\log_2\kappa-1,\ldots,0\)}{
        Reconstruct \(\bm C_\tau(Z)\) and \(\bm C_{\tau+1}(Z)\)
        from the witness, reusing the coefficient workspace\;
        Form \(\bm g_\tau(Z,X)\leftarrow\bm C_\tau(Z)\bm\psi(X)\)\;
        Aggregate each column of \(\bm C_{\tau+1}(Z)\) against
        \(\ext{\bm l}_{\tau+1}\), then evaluate at \(\ext z\) to obtain
        \(\ext{\bm u}_{\tau+1}^\top\)\;
        Set \(g_C(X)\leftarrow\ext{\bm u}_{\tau+1}^\top\bm\psi(X)\)\;
        \For{\(j=1,\ldots,\kappa-1\)}{
            Compute
            \(\bm\eta_{\tau+1,j}(Z)
            \leftarrow\bm g_{\tau,U}(Z,\gamma'_j)
                       \circ\bm g_{\tau,D}(Z,\gamma'_j)
            =\sum_{\nu=0}^{2^{\tau+1}}\bm p_{\tau+1,j,\nu}Z^\nu\)
            over the interpolation field\;
            Set \(F_{\tau+1,j}(Z)
            \leftarrow\sum_{\nu=0}^{2^{\tau+1}}
            \langle\ext{\bm l}_{\tau+1},\bm p_{\tau+1,j,\nu}\rangle Z^\nu\),
            then evaluate \(F_{\tau+1,j}(\ext z)\)\;
        }
        Interpolate \(h(X)\) of degree at most \(\kappa-2\) from
        \(h(\gamma'_j)=
        (F_{\tau+1,j}(\ext z)-g_C(\gamma'_j))/\Psi(\gamma'_j)\)\;
        \(\PtoV:\ext{\bm u}_{\tau+1},h\);
        check \(\ext v_{\tau+1}
        =\ext{\bm u}_{\tau+1}^\top\ext{\bm r}_{\tau+1}\)\;
        Sample \(\VtoP:\ext x\ChooseRandom\ExtensionField\setminus\mathcal H\)
        and set \(\ext{\bm\rho}(\ext x)\leftarrow\bm\psi(\ext x)\)\;
        Apply cubic weighted sumcheck to
        \(\langle\ext{\bm l}_{\tau+1},
        \bm g_{\tau,U}(\ext z,\ext x)\circ\bm g_{\tau,D}(\ext z,\ext x)\rangle
        =g_C(\ext x)+\Psi(\ext x)h(\ext x)\),
        obtaining child values \(\ext v_{\tau,U},\ext v_{\tau,D}\)
        at the same row MLE point \(\ext{\bm y}\)\;
        After both values are fixed, sample
        \(\VtoP:\ext y_0\ChooseRandom\ExtensionField\) and set
        \(\ext v_\tau\leftarrow
        (1-\ext y_0)\ext v_{\tau,U}+\ext y_0\ext v_{\tau,D}\),
        \(\ext{\bm l}_\tau\leftarrow
        \tensorspan{\ext y_0\mathbin\|\ext{\bm y}}\), and
        \(\ext{\bm r}_\tau\leftarrow\ext{\bm\rho}(\ext x)\)\;
    }
    Output \(\ext v_0-\ext z
    =\ext{\bm l}_0^\top\bm D\ext{\bm r}_0\)
    for authentication by Section~\ref{sec:pcs}\;
}
\Return the conjunction of all reduction checks and the two terminal claims\;
\end{algorithm}
